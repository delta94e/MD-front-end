### 📖 Hidden Classes và Shape Optimization

V8 engine sử dụng "Hidden Classes" (hay "Shapes") để optimize property access:



**Hidden Class Creation Process:**

### 📖 Object State Debugging Strategies


#### State Diff Implementation


```javascript
// Netflix video player state debugging
function createStateDiffer() {
  let previousState = null;

  return function diff(currentState) {
    if (!previousState) {
      previousState = JSON.parse(JSON.stringify(currentState));
      return { type: 'initial', state: currentState };
    }

    const changes = findChanges(previousState, currentState);
    previousState = JSON.parse(JSON.stringify(currentState));

    return { type: 'update', changes };
  };
}

function findChanges(prev, curr, path = '') {
  const changes = [];

  for (const key in curr) {
    const currentPath = path ? `${path}.${key}` : key;

    if (!(key in prev)) {
      changes.push({ type: 'added', path: currentPath, value: curr[key] });
    } else if (prev[key] !== curr[key]) {
      if (typeof curr[key] === 'object' && curr[key] !== null) {
        changes.push(...findChanges(prev[key], curr[key], currentPath));
      } else {
        changes.push({
          type: 'changed',
          path: currentPath,
          oldValue: prev[key],
          newValue: curr[key]
        });
      }
    }
  }

  for (const key in prev) {
    if (!(key in curr)) {
      const currentPath = path ? `${path}.${key}` : key;
      changes.push({ type: 'removed', path: currentPath, oldValue: prev[key] });
    }
  }

  return changes;
}
```

```javascript
// Facebook privacy settings implementation
function createSecureUser(userData, permissions) {
  const privateProperties = new Set(['ssn', 'creditCard', 'internalId']);

  return new Proxy(userData, {
    get(target, prop) {
      if (privateProperties.has(prop) && !permissions.canAccessPrivate) {
        throw new Error(`Access denied to property: ${prop}`);
      }
      return target[prop];
    },

    set(target, prop, value) {
      if (privateProperties.has(prop) && !permissions.canModifyPrivate) {
        throw new Error(`Modification denied for property: ${prop}`);
      }
      target[prop] = value;
      return true;
    },

    has(target, prop) {
      if (privateProperties.has(prop) && !permissions.canAccessPrivate) {
        return false; // Hide existence of private properties
      }
      return prop in target;
    },

    ownKeys(target) {
      const keys = Object.keys(target);
      if (!permissions.canAccessPrivate) {
        return keys.filter(key => !privateProperties.has(key));
      }
      return keys;
    }
  });
}
```

### 📖 Observer Pattern Implementation


#### Event-Driven Object Architecture


```javascript
// Netflix video player event system
class EventDrivenObject {
  constructor() {
    this._listeners = new Map();
    this._properties = new Map();

    return new Proxy(this, {
      get(target, prop) {
        if (prop.startsWith('_') || typeof target[prop] === 'function') {
          return target[prop];
        }

        target._trackAccess(prop);
        return target._properties.get(prop);
      },

      set(target, prop, value) {
        if (prop.startsWith('_')) {
          target[prop] = value;
          return true;
        }

        const oldValue = target._properties.get(prop);
        target._properties.set(prop, value);
        target._notifyChange(prop, oldValue, value);
        return true;
      }
    });
  }

  on(property, callback) {
    if (!this._listeners.has(property)) {
      this._listeners.set(property, new Set());
    }
    this._listeners.get(property).add(callback);
  }

  off(property, callback) {
    if (this._listeners.has(property)) {
      this._listeners.get(property).delete(callback);
    }
  }

  _notifyChange(property, oldValue, newValue) {
    if (this._listeners.has(property)) {
      this._listeners.get(property).forEach(callback => {
        callback(newValue, oldValue, property);
      });
    }
  }

  _trackAccess(property) {
    // Analytics tracking for property access patterns
    if (this._analytics) {
      this._analytics.trackPropertyAccess(property);
    }
  }
}
```


### 📖 State Machine Pattern

```javascript
// Amazon order state machine
class OrderStateMachine {
  constructor(initialState = 'pending') {
    this.state = initialState;
    this.transitions = new Map([
      ['pending', new Set(['confirmed', 'cancelled'])],
      ['confirmed', new Set(['shipped', 'cancelled'])],
      ['shipped', new Set(['delivered', 'returned'])],
      ['delivered', new Set(['returned'])],
      ['cancelled', new Set()],
      ['returned', new Set()]
    ]);

    this.stateHandlers = new Map([
      ['confirmed', () => this._processPayment()],
      ['shipped', () => this._notifyShipping()],
      ['delivered', () => this._requestReview()],
      ['cancelled', () => this._refundPayment()],
      ['returned', () => this._processReturn()]
    ]);
  }

  transition(newState) {
    const validTransitions = this.transitions.get(this.state);

    if (!validTransitions || !validTransitions.has(newState)) {
      throw new Error(`Invalid transition from ${this.state} to ${newState}`);
    }
```

### 📖 Object Pooling Strategies


#### High-Performance Object Reuse


```javascript
// Netflix video thumbnail pool
class ObjectPool {
  constructor(createFn, resetFn, maxSize = 100) {
    this.createFn = createFn;
    this.resetFn = resetFn;
    this.maxSize = maxSize;
    this.pool = [];
    this.activeObjects = new Set();
  }

  acquire() {
    let obj;

    if (this.pool.length > 0) {
      obj = this.pool.pop();
    } else {
      obj = this.createFn();
    }

    this.activeObjects.add(obj);
    return obj;
  }

  release(obj) {
    if (!this.activeObjects.has(obj)) {
      throw new Error('Object not from this pool');
    }

    this.activeObjects.delete(obj);

    if (this.pool.length < this.maxSize) {
      this.resetFn(obj);
      this.pool.push(obj);
    }
    // Objects beyond maxSize are abandoned for GC
  }

  // Metrics for monitoring
  getStats() {
    return {
      poolSize: this.pool.length,
      activeCount: this.activeObjects.size,
      totalCapacity: this.maxSize
    };
  }
}

    obj.width = null;
    obj.height = null;
```

## Phần I: Foundation Level - Memory Model và Reference Fundamentals


### 📖 Tại Sao Objects Khác Fundamentally Với Primitives?


#### 🌱 Nguồn Gốc & Computer Science Foundation


Trước khi hiểu object references, chúng ta cần understand memory model cơ bản của computer systems. Điều này connect trực tiếp với computer science fundamentals.


**Memory Architecture Overview:**


Trong computer memory, có hai regions chính:


1. **Stack Memory**: Fast access, limited size, stores primitives và references
2. **Heap Memory**: Slower access, larger size, stores actual objects


```javascript
// Primitive - stored directly trong stack
let age = 30;        // Stack: [age: 30]
let name = "John";   // Stack: [name: "John"]

// Object - reference trong stack, data trong heap
let user = { name: "John", age: 30 };
// Stack: [user: <memory_address_0x1234>]
// Heap: [0x1234: {name: "John", age: 30}]
```


#### 🔬 Tại Sao Cần Architecture Này?


**Performance Optimization Reasons:**


1. **Stack Access Speed**: CPU có thể access stack data trong 1-2 cycles
2. **Memory Efficiency**: Objects có thể có variable size, heap handles dynamic allocation better
3. **Garbage Collection**: Heap allows sophisticated GC algorithms
4. **Sharing**: Multiple references có thể point đến same object without duplication


#### 💭 Principal's Deep Understanding


Khi tôi first time gặp concept này năm 2012, biggest confusion của tôi là: "Tại sao JavaScript không cho phép access trực tiếp địa chỉ memory như C/C++?"


Answer: JavaScript abstracts memory management để prevent security vulnerabilities và memory leaks. Developer work với references một cách safe, trong khi engine handle actual memory operations.


### 📖 Primitive vs Object Copying: Chi Tiết Mechanism


#### Primitive Copying Deep Dive


```javascript
let message = "Hello!";
let phrase = message;
```


**Step-by-step Execution:**


1. **Memory Allocation**: Engine allocate stack space cho `message`
2. **Value Storage**: String "Hello!" được store directly trong `message`'s memory slot
3. **Copy Operation**: Khi assign `message` to `phrase`, engine creates NEW memory slot
4. **Independent Values**: `message` và `phrase` now completely independent


```
Stack Memory Layout:
[message: "Hello!"]  <- Address 0x1000
[phrase:  "Hello!"]  <- Address 0x1004 (different location)
```


#### Object Reference Copying Analysis


```javascript
let user = { name: "John" };
let admin = user;
```


**Detailed Execution Process:**


1. **Object Creation**: Engine allocate heap memory cho object `{ name: "John" }`
2. **Reference Storage**: Variable `user` stores heap address, NOT object itself
3. **Reference Copy**: Assignment copies ADDRESS, not object content
4. **Shared Access**: Both variables now reference same heap location


```
Memory Layout:
Stack:
[user:  0x2000]  <- Points to heap
[admin: 0x2000]  <- Same address!

Heap:
[0x2000: { name: "John" }]  <- Single object
```

```javascript
function objectAssign(target, ...sources) {

  const to = Object(target);

  sources.forEach(source => {
```

#### Object Pooling Pattern


Netflix video thumbnail pooling implementation:


```javascript
class ObjectPool {
  constructor(createFn, resetFn, maxSize = 100) {
    this.createFn = createFn;
    this.resetFn = resetFn;
    this.maxSize = maxSize;
    this.available = [];
    this.inUse = new Set();
    this.totalCreated = 0;
  }

  acquire() {
    let obj;

    if (this.available.length > 0) {
      obj = this.available.pop();
    } else {
      obj = this.createFn();
      this.totalCreated++;
    }

    this.inUse.add(obj);
    return obj;
  }

  release(obj) {
    if (!this.inUse.has(obj)) {
      throw new Error('Object not from this pool');
    }

    this.inUse.delete(obj);

    if (this.available.length < this.maxSize) {
      this.resetFn(obj);
      this.available.push(obj);
    }
    // Allow GC for objects beyond pool size
  }

  getMetrics() {
    return {
      available: this.available.length,
      inUse: this.inUse.size,
      totalCreated: this.totalCreated,
      hitRate: (this.totalCreated - this.inUse.size) / this.totalCreated
    };
  }
}

    obj.width = null;
    obj.height = null;
    obj.loaded = false;
    if (obj.element) {
      obj.element.src = '';
```

```javascript
class ImmutableState {
  constructor(initialState) {
    this.state = initialState;
    this.frozen = false;
  }

  // Lazy freezing - only freeze when accessed
  get() {
    if (!this.frozen) {
      this.state = this.deepFreeze(this.state);
      this.frozen = true;
    }
    return this.state;
  }

  // Copy-on-write mutation
  update(path, value) {
    if (this.frozen) {
      // Create new state tree with minimal copying
      this.state = this.updatePath(this.state, path, value);
      this.frozen = false;
    } else {
      // Direct mutation is safe if not frozen yet
      this.setPath(this.state, path, value);
    }
    return this;
  }

  updatePath(obj, path, value) {
    if (path.length === 0) return value;

    const [head, ...tail] = path;
    const newObj = Array.isArray(obj) ? [...obj] : { ...obj };
    newObj[head] = this.updatePath(obj[head] || {}, tail, value);
    return newObj;
  }

  deepFreeze(obj) {
    Object.getOwnPropertyNames(obj).forEach(prop => {
      const value = obj[prop];
      if (value && typeof value === 'object') {
        this.deepFreeze(value);
      }
    });
    return Object.freeze(obj);
  }
}

    this.state = new ImmutableState({
      user: { name: 'John', preferences: { theme: 'dark' } },
      posts: []
    });
  }

  updateUserName(newName) {
    this.setState(
      this.state.update(['user', 'name'], newName).get()
    );
```

```javascript
class SmartReference {
  constructor(target, options = {}) {
    this.target = target;
    this.options = options;
    this.accessLog = [];
    this.mutationLog = [];
    this.listeners = new Map();

    return new Proxy(target, {
      get: (obj, prop, receiver) => {
        this.trackAccess(prop);
        this.notifyListeners('get', { prop, value: obj[prop] });
        return Reflect.get(obj, prop, receiver);
      },

      set: (obj, prop, value, receiver) => {
        const oldValue = obj[prop];
        this.trackMutation(prop, oldValue, value);
        this.notifyListeners('set', { prop, oldValue, newValue: value });
        return Reflect.set(obj, prop, value, receiver);
      },

      deleteProperty: (obj, prop) => {
        const oldValue = obj[prop];
        this.trackMutation(prop, oldValue, undefined);
        this.notifyListeners('delete', { prop, oldValue });
        return Reflect.deleteProperty(obj, prop);
      }
    });
  }

  trackAccess(prop) {
    if (this.options.logAccess) {
      this.accessLog.push({
        property: prop,
        timestamp: performance.now(),
        stack: this.options.captureStack ? new Error().stack : null
      });
    }
  }

  trackMutation(prop, oldValue, newValue) {
    if (this.options.logMutations) {
      this.mutationLog.push({
        property: prop,
        oldValue,
        newValue,
        timestamp: performance.now(),
        stack: this.options.captureStack ? new Error().stack : null
      });
    }
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, new Set());
    }
    this.listeners.get(event).add(callback);
  }

  notifyListeners(event, data) {
    const eventListeners = this.listeners.get(event);
    if (eventListeners) {
      eventListeners.forEach(callback => {
        try {
          callback(data);
        } catch (e) {
          console.error('Listener error:', e);
        }
      });
    }
  }
}

    logMutations: true,
```

```javascript
class MemoryTracker {
  constructor() {
    this.snapshots = [];
    this.objectCounts = new Map();
    this.leakDetection = new WeakSet();
  }

  takeSnapshot() {
    const snapshot = {
      timestamp: Date.now(),
      heapUsed: this.getHeapUsage(),
      objectCounts: new Map(this.objectCounts)
    };

    this.snapshots.push(snapshot);

    // Keep only last 100 snapshots
    if (this.snapshots.length > 100) {
      this.snapshots.shift();
    }

    return snapshot;
  }

  trackObject(obj, type) {
    const count = this.objectCounts.get(type) || 0;
    this.objectCounts.set(type, count + 1);

    // Add to leak detection set
    this.leakDetection.add(obj);

    // Schedule cleanup check
    setTimeout(() => {
      if (this.leakDetection.has(obj)) {
        console.warn(`Potential memory leak detected for ${type}`);
      }
    }, 60000); // Check after 1 minute
  }

  getHeapUsage() {
    if (performance.memory) {
      return {
        used: performance.memory.usedJSHeapSize,
        total: performance.memory.totalJSHeapSize,
        limit: performance.memory.jsHeapSizeLimit
      };
    }
    return null;
  }

  analyzeLeaks() {
    const recent = this.snapshots.slice(-10);
    if (recent.length < 2) return null;

    const growth = recent.map((snapshot, i) => {
      if (i === 0) return 0;
      return snapshot.heapUsed.used - recent[i-1].heapUsed.used;
    });

    const avgGrowth = growth.reduce((a, b) => a + b, 0) / growth.length;

    return {
      averageGrowth: avgGrowth,
      totalGrowth: recent[recent.length-1].heapUsed.used - recent[0].heapUsed.used,
      isLeaking: avgGrowth > 1024 * 1024 // 1MB average growth
    };
  }
}
```

## Phần I: Foundation Level - Memory Management Fundamentals


### 📖 Reachability: The Core Concept


#### 🌱 Nguồn Gốc & Computer Science Foundation


Trước khi deep dive vào JavaScript specifics, chúng ta cần understand fundamental problem mà garbage collection solve: **automatic memory management**.


**Historical Context:**


Trong early programming languages như C/C++, developers phải manually manage memory:


```c
// Manual memory management - C style
int* data = malloc(sizeof(int) * 1000);  // Allocate
// Use data...
free(data);  // Must remember to free!
```


Problems với manual memory management:


1. **Memory Leaks**: Forget to free → memory never returned
2. **Dangling Pointers**: Use freed memory → crashes
3. **Double Free**: Free same memory twice → corruption
4. **Cognitive Overhead**: Developer must track all allocations


**The Solution: Automatic Memory Management**


JavaScript adopts automatic memory management approach. Core principle: "If memory cannot be reached từ program execution, it's safe to free."


#### 🔬 Reachability Definition Deep Dive


**Formal Definition:**


>
> A value is reachable if it can be accessed or used by the currently executing program through any chain of references starting from root objects.
>
>


**Root Objects (GC Roots):**
Những objects mà garbage collector never xóa because they're inherently needed:


1. **Global Variables**


```javascript
var globalUser = { name: "John" };  // Always reachable
```


1. **Currently Executing Function Scope**


```javascript
function processUser() {
    let localUser = { name: "Jane" };  // Reachable while function executing
    // localUser becomes unreachable when function exits
}
```


1. **Call Stack Variables**


```javascript
function outer() {
    let outerVar = { data: "important" };

    function inner() {
        console.log(outerVar.data);  // outerVar reachable through closure
    }

    return inner;  // outerVar stays reachable through returned function
}
```


1. **DOM References (Browser Environment)**


```javascript
let element = document.getElementById('myDiv');  // DOM node reference
```


#### 💭 Principal's Mental Model


Khi tôi first learned về reachability, biggest confusion là: "Tại sao JavaScript engine không just track mọi object creation và destruction?"


Answer: Performance và complexity. Tracking every object individually would require massive overhead. Reachability-based approach cho phép engine focus only on objects that matter, batch cleanup operations, và optimize cho common usage patterns.


Think của reachability như "breadcrumb trail". Nếu bạn có thể follow trail từ current execution context đến object, object đó reachable. Nếu trail bị broken, object becomes unreachable.


### 📖 Simple Example Deep Analysis


#### Basic Reference Management


```javascript
// user has a reference to the object
let user = {
    name: "John"
};
```


**Memory Layout Analysis:**


```
Stack Memory:
[user: 0x1000] <- Variable points to heap address

Heap Memory:
[0x1000: { name: "John" }] <- Actual object data
```


**When assignment happens:**


```javascript
user = null;
```


**Memory Changes:**


```
Stack Memory:
[user: null] <- No longer points to object

Heap Memory:
[0x1000: { name: "John" }] <- Object still exists but unreachable
```


**Garbage Collection Result:**
Object tại 0x1000 becomes eligible for collection vì no references exist.


#### 🔬 Step-by-step Garbage Collection Process


**Phase 1: Mark Phase**


1. GC starts từ root objects
2. Follows all references recursively
3. Marks every reachable object
4. Creates "marked set" of live objects


**Phase 2: Sweep Phase**


1. Scans entire heap memory
2. Any object not trong "marked set" = garbage
3. Frees memory occupied by garbage objects
4. Updates memory allocation pointers


**Phase 3: Compact Phase (Optional)**


1. Moves live objects together
2. Reduces memory fragmentation
3. Updates all references to moved objects

### 📖 Two References Pattern Analysis


#### Reference Sharing Deep Dive


```javascript
// user has a reference to the object
let user = {
    name: "John"
};

let admin = user;  // Copy reference, NOT object
```


**Critical Understanding:**


- Variables `user` và `admin` are NOT copies
- They're two different pointers to SAME memory location
- Object exists once trong heap, referenced twice từ stack


**Memory Model:**


```
Stack Memory:
[user:  0x2000] <- Points to heap
[admin: 0x2000] <- Points to SAME heap location

Heap Memory:
[0x2000: { name: "John" }] <- Single object, multiple references
```


#### Reference Counting vs Mark-and-Sweep


**Reference Counting (Old Approach):**


- Track number of references to each object
- When count reaches 0, immediately free object
- Problem: Circular references never get count = 0


**Mark-and-Sweep (Modern Approach):**


- Don't count references
- Find all reachable objects từ roots
- Free everything else
- Handles circular references correctly

- Ann reachable through: `family.mother` AND `family.father.wife`


#### 🔬 Partial Reference Removal Analysis


```javascript
delete family.father;
```

**After First Deletion (delete family.father):**
John object still reachable through:


- `family.mother.husband`


**After Second Deletion (delete family.mother.husband):**
John object no longer reachable through any path từ `family` root.


**Critical Insight:**
Object reachability depends on ALL possible paths từ roots. Removing some references doesn't guarantee unreachability.


#### Production Pattern: Event System Cleanup


Google Analytics implementation pattern:

```javascript
        this.events = [];
        this.listeners = new Map();

        const handler = (e) => {
            this.recordEvent(eventName, e);
        };

        // CREATE CIRCULAR REFERENCE
        const eventObj = {
            name: eventName,
            element: element,
            handler: handler,

            event.element.removeEventListener('click', event.handler);
            event.tracker = null;
            event.element = null;
            event.handler = null;
        });

        this.events.length = 0;
        this.listeners.clear();
    }
```

### 📖 Unreachable Islands: Advanced Pattern


#### Island Formation Mechanism


```javascript
let family = marry(john, ann);

```


**Before family = null:**


```
Root Reference: family -> main object
Main object connects to circular graph:
John ←→ Ann (circular references)
```


**After family = null:**


```
No root references to any part of graph
John ←→ Ann still reference each other
BUT: Entire graph unreachable từ any root
```


**Garbage Collection Result:**
Entire "island" of objects gets collected, even though internal references exist.


#### 💭 Advanced Mental Model


Think của unreachable islands như isolated networks. Network có thể có complex internal connectivity, nhưng if no connection to outside world exists, entire network can be shut down.


This is fundamental difference between:


- **Reference Counting**: Would never free circular references
- **Mark-and-Sweep**: Correctly identifies unreachable islands

#### 🔬 Mark Phase Implementation


**Conceptual Implementation:**


```javascript
// Simplified mark-and-sweep implementation
class GarbageCollector {
    constructor() {
        this.heap = new Set();        // All allocated objects
        this.marked = new Set();      // Objects marked as reachable
        this.roots = new Set();       // Root objects
    }

    // Phase 1: Mark all reachable objects
    markPhase() {
        this.marked.clear();

        // Start từ all roots
        this.roots.forEach(root => {
            this.markObject(root);
        });
    }

    markObject(obj) {
        // Avoid infinite loops với circular references
        if (this.marked.has(obj)) {
            return;
        }

        // Mark current object
        this.marked.add(obj);

        // Recursively mark all referenced objects
        this.getReferences(obj).forEach(ref => {
            this.markObject(ref);
        });
    }

    // Phase 2: Sweep unmarked objects
    sweepPhase() {
        const garbage = [];

        this.heap.forEach(obj => {
            if (!this.marked.has(obj)) {
                garbage.push(obj);
            }
        });

        // Free garbage objects
        garbage.forEach(obj => {
            this.freeObject(obj);
            this.heap.delete(obj);
        });

        return garbage.length;
    }
}
```

**Incremental Collection:**

```javascript
// Conceptual incremental marking
class IncrementalGC {
    constructor() {
        this.markingQueue = [];
        this.maxMarkingTime = 5; // ms per frame
    }

    performIncrementalMark() {
        const startTime = performance.now();

        while (this.markingQueue.length > 0 &&
               (performance.now() - startTime) < this.maxMarkingTime) {

            const obj = this.markingQueue.shift();
            this.markObject(obj);
        }

        // If more work needed, schedule continuation
        if (this.markingQueue.length > 0) {
            requestIdleCallback(() => this.performIncrementalMark());
        }
    }
}
```

```javascript
    constructor() {
        this.observers = new WeakMap();
        this.elementObservers = new WeakMap();
    }

    observe(element, callback) {
        if (!this.elementObservers.has(element)) {
            this.elementObservers.set(element, new Set());
        }

        this.elementObservers.get(element).add(callback);

        // Automatic cleanup khi element removed từ DOM
        const observer = new MutationObserver((mutations) => {
            mutations.forEach(mutation => {
                mutation.removedNodes.forEach(node => {
                    if (node === element) {
                        this.cleanup(element);
                    }
                });
            });
        });

        observer.observe(document.body, {
            childList: true,
            subtree: true
        });

        this.observers.set(element, observer);
    }

    cleanup(element) {
        const observer = this.observers.get(element);
        if (observer) {
            observer.disconnect();
        }
        // WeakMap entries automatically cleaned up
```

```javascript
            if (i === 0) return 0;
            return snapshot.used - recent[i-1].used;

        // Alert if consistent memory growth
        if (avgGrowth > 1024 * 1024) { // 1MB average growth
            this.alertMemoryLeak(avgGrowth);

                type: 'memory_leak',
                growthRate,
                timestamp: Date.now(),
                userAgent: navigator.userAgent
            })
        });
    }
```

#### Object Pooling Pattern


```javascript
// Netflix video thumbnail pooling
class ThumbnailPool {
    constructor(maxSize = 200) {
        this.pool = [];
        this.inUse = new Set();
        this.maxSize = maxSize;
        this.createCount = 0;
    }

    acquire() {
        let thumbnail;

        if (this.pool.length > 0) {
            thumbnail = this.pool.pop();
        } else {
            thumbnail = this.createThumbnail();
            this.createCount++;
        }

        this.inUse.add(thumbnail);
        return thumbnail;
    }

    release(thumbnail) {
        if (!this.inUse.has(thumbnail)) {
            console.warn('Thumbnail not từ this pool');
            return;
        }

        this.inUse.delete(thumbnail);

        if (this.pool.length < this.maxSize) {
            this.resetThumbnail(thumbnail);
            this.pool.push(thumbnail);
        }
        // Objects beyond maxSize allowed to be GC'd
    }

    createThumbnail() {
        return {
            element: document.createElement('img'),
            data: null,
            loaded: false,
            clickHandler: null
        };
    }

    resetThumbnail(thumbnail) {
        thumbnail.element.src = '';
        thumbnail.data = null;
        thumbnail.loaded = false;

        if (thumbnail.clickHandler) {
            thumbnail.element.removeEventListener('click', thumbnail.clickHandler);
            thumbnail.clickHandler = null;
        }
    }

    getStats() {
        return {
            poolSize: this.pool.length,
            inUse: this.inUse.size,
            totalCreated: this.createCount,
            hitRate: (this.createCount - this.inUse.size) / this.createCount
        };
    }
}

        return thumbnail;
```

```javascript
            .then(data => {

                id: data.id,
                title: data.title,
                price: data.price
            };

            this.loadedData.set(element, minimalData);
        }
    }
```

### 📖 Nguồn Gốc và Motivation: Tại Sao Methods Tồn Tại?


#### 🌱 Historical Context và Problem Statement


Trước khi hiểu object methods, chúng ta cần understand fundamental problem mà object-oriented programming solve: **data và behavior encapsulation**.


**Early Programming Model:**
Trong procedural programming languages như C, data và functions completely separated:


```c
// C-style programming - data và functions riêng biệt
struct User {
    char name[50];
    int age;
};

void sayHello(struct User* user) {  // Function operates on data
    printf("Hello, %s!", user->name);
}

// Usage
struct User john = {"John", 30};
sayHello(&john);  // Pass data to function
```


**Problems với Approach Này:**


1. **No Encapsulation**: Data và behavior không connected logically
2. **Namespace Pollution**: Functions scattered across global namespace
3. **No Data Hiding**: All struct members publicly accessible
4. **Maintenance Nightmare**: Related functions không grouped together
5. **Code Reusability**: Difficult to create reusable components


**The Object-Oriented Solution:**
JavaScript adopts object-oriented approach where data (properties) và behavior (methods) được grouped together:


```javascript
// JavaScript object - data và behavior encapsulated
let user = {
    // Data (properties)
    name: "John",
    age: 30,

    // Behavior (methods)
    sayHello: function() {
        console.log("Hello, " + this.name + "!");
    }
};

// Usage
user.sayHello();  // Object performs action on itself
```

Objects implement **Abstract Data Type (ADT)** concept từ computer science:


**ADT Principles:**


1. **Data Abstraction**: Hide internal implementation details
2. **Encapsulation**: Group related data và operations together
3. **Interface**: Consistent way to interact với object
4. **Information Hiding**: Control access to internal state

### 📖 "This" Keyword: The Heart of Object-Oriented JavaScript


#### 🌱 Fundamental Understanding: What is "This"?


**Definition:**
`this` keyword trong JavaScript refers to the object that is executing the current function. Critical insight: `this` is determined by **how function is called**, not where it's defined.


**Computer Science Background:**
`this` implements **dynamic binding** concept từ OOP theory:


- **Static Binding**: Method binding determined at compile time
- **Dynamic Binding**: Method binding determined at runtime


JavaScript uses dynamic binding để enable flexible, reusable code patterns.


#### 🔬 "This" Binding Rules: Complete Technical Analysis


**Rule 1: Method Invocation Binding**


```javascript
let user = {
    name: "John",
    sayHi() {
        console.log(this.name); // `this` = user object
    }
};

user.sayHi(); // Method invocation - `this` bound to user
```


**Execution Flow:**


1. **Property Access**: `user.sayHi` retrieves function reference
2. **Call Site Analysis**: Engine detects method call pattern `object.method()`
3. **This Binding**: Engine sets `this` to object before dot (user)
4. **Function Execution**: Function runs với `this` = user


**Rule 2: Function Invocation Binding**


```javascript
function sayHi() {
    console.log(this); // `this` = undefined (strict mode) or global object
}

sayHi(); // Function invocation - no implicit binding
```


**Strict Mode vs Non-Strict Mode:**


```javascript
// Strict mode
"use strict";
function test() {
    console.log(this); // undefined
}
test();

// Non-strict mode
function test() {
    console.log(this); // window object (browsers) or global (Node.js)
}
test();
```


**Rule 3: Call/Apply/Bind Explicit Binding**


```javascript
let user = { name: "John" };
let admin = { name: "Admin" };

function greet() {
    console.log("Hello, " + this.name);
}

greet.call(user);  // Explicit binding - `this` = user
greet.call(admin); // Explicit binding - `this` = admin
```


**Rule 4: Arrow Function Lexical Binding**


```javascript
let user = {
    name: "John",
    sayHi() {
        let arrow = () => {
            console.log(this.name); // `this` inherited từ outer function
        };
        arrow();
    }
};

user.sayHi(); // `this` trong arrow function = user
```


#### Step-by-step "This" Resolution Algorithm


**JavaScript Engine's This Resolution Process:**


```javascript
function determineThis(callSite, functionRef) {
    // Step 1: Check for explicit binding (call/apply/bind)
    if (hasExplicitBinding(callSite)) {
        return getExplicitBinding(callSite);
    }

    // Step 2: Check for implicit binding (method call)
    if (hasImplicitBinding(callSite)) {
        return getContextObject(callSite);
    }

    // Step 3: Check for arrow function
    if (isArrowFunction(functionRef)) {
        return getLexicalThis(functionRef);
    }

    // Step 4: Default binding
    if (isStrictMode()) {
        return undefined;
    } else {
        return globalObject;
    }
}
```

```javascript
        return this;
```

```javascript
        return this;
```

```javascript
        return subtotal - discountAmount + tax + shipping;
```

```javascript
class RouteBuilder {
    constructor() {
        this.waypoints = [];
        this.options = {
            avoidTolls: false,
            avoidHighways: false,
            optimize: false
        };
        this.errors = [];
    }

            this.waypoints.push(location);
        } else {
            this.errors.push(`Invalid location: ${location}`);
        }
        return this;
    }

    // Conditional chaining - only execute if condition met
    when(condition, callback) {
        if (condition) {
            callback.call(this, this);
        }
        return this;
    }

    unless(condition, callback) {
        if (!condition) {
            callback.call(this, this);
        }
        return this;
    }

    avoidTolls(avoid = true) {
        this.options.avoidTolls = avoid;
        return this;
    }

    avoidHighways(avoid = true) {
        this.options.avoidHighways = avoid;
        return this;
    }

    optimize(enable = true) {
        this.options.optimize = enable;
        return this;
    }

        // Validation logic
        return location && typeof location === 'string' && location.length > 0;

        if (this.errors.length > 0) {
            throw new Error(`Route building failed: ${this.errors.join(', ')}`);
        }

        if (this.waypoints.length < 2) {
            throw new Error('Route requires at least 2 waypoints');
```

```javascript
            // Step 3: Shipping arrangement
            await this.executeWithRetry(

            );

            // Step 4: Order confirmation
            await this.confirmOrder(orderSession);

            return {
                success: true,
                orderId: orderSession.orderId,
                trackingNumber: orderSession.trackingNumber
            };

        } catch (error) {
            // Comprehensive rollback
            await this.rollbackOrder(orderSession, error);
            throw error;
        }
    }

    // Retry mechanism với exponential backoff
    async executeWithRetry(operation, operationName) {
        let lastError;

        for (let attempt = 1; attempt <= this.retryConfig.maxRetries; attempt++) {
            try {
                return await operation();
            } catch (error) {
                lastError = error;

                if (!this.isRetryableError(error) || attempt === this.retryConfig.maxRetries) {
                    throw error;
                }

                // Exponential backoff delay
                const delay = this.retryConfig.baseDelay *
                            Math.pow(this.retryConfig.backoffMultiplier, attempt - 1);

                console.warn(`${operationName} failed (attempt ${attempt}), retrying in ${delay}ms`);
                await this.sleep(delay);
            }
        }

        throw lastError;
    }

    // Error classification
    isRetryableError(error) {
        const retryableErrors = [
            'NETWORK_TIMEOUT',
            'SERVICE_UNAVAILABLE',
            'RATE_LIMIT_EXCEEDED',
            'TEMPORARY_FAILURE'
        ];

        return retryableErrors.includes(error.code);
    }

    // Rollback coordination
    async rollbackOrder(orderSession, originalError) {
        const rollbackOperations = [];

        if (orderSession.shippingArranged) {
            rollbackOperations.push(
                () => this.shippingService.cancelShipping(orderSession.shippingId)
            );
        }

        if (orderSession.paymentProcessed) {
            rollbackOperations.push(
                () => this.paymentService.refundPayment(orderSession.paymentId)
            );
        }

        if (orderSession.inventoryReserved) {
            rollbackOperations.push(
                () => this.inventoryService.releaseReservation(orderSession.reservationId)
            );
        }

        // Execute rollbacks in parallel
        const rollbackPromises = rollbackOperations.map(async (operation) => {
            try {
                await operation();
            } catch (rollbackError) {
                console.error('Rollback operation failed:', rollbackError);
                // Log but don't throw - continue với other rollbacks
            }
        });

        await Promise.allSettled(rollbackPromises);

        // Notify monitoring systems
        this.notificationService.notifyOrderFailure(orderSession, originalError);
    }

        return {
            orderId: this.generateOrderId(),
            orderData,
            timestamp: Date.now(),
            inventoryReserved: false,
            paymentProcessed: false,
```

```javascript
class MethodPerformanceMonitor {
    constructor() {
        this.metrics = new Map();
        this.ongoingCalls = new Map();
    }

    trackMethod(methodName, callback) {
        const callId = this.generateCallId();
        const startTime = performance.now();

        this.ongoingCalls.set(callId, { methodName, startTime });

        try {
            const result = callback();

            if (result instanceof Promise) {
                return result.finally(() => {
                    this.recordMetric(callId);
                });
            } else {
                this.recordMetric(callId);
                return result;
            }
        } catch (error) {
            this.recordMetric(callId, error);
            throw error;
        }
    }

    recordMetric(callId, error = null) {
        const callData = this.ongoingCalls.get(callId);
        if (!callData) return;

        const endTime = performance.now();
        const duration = endTime - callData.startTime;

        if (!this.metrics.has(callData.methodName)) {
            this.metrics.set(callData.methodName, {
                calls: 0,
                totalDuration: 0,
                errors: 0,
                avgDuration: 0
            });
        }

        const metric = this.metrics.get(callData.methodName);
        metric.calls++;
        metric.totalDuration += duration;
        metric.avgDuration = metric.totalDuration / metric.calls;

        if (error) {
            metric.errors++;
        }

        this.ongoingCalls.delete(callId);
    }

        return `call_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
```

```javascript
                    const getNameRegular = function() {
                        return this.name;
                    };

                    return { getNameArrow, getNameRegular };
                }
            };

            const { getNameArrow, getNameRegular } = component.init();

            expect(getNameArrow()).toBe('TestComponent');
            expect(() => getNameRegular()).toThrow();
        });
    });

    describe('Complex method scenarios', () => {
        it('should handle method borrowing correctly', () => {
            const user1 = { name: 'John' };
            const user2 = { name: 'Jane' };

            const greet = function() {
                return `Hello, ${this.name}!`;
            };

            user1.greet = greet;
            user2.greet = greet;

            expect(user1.greet()).toBe('Hello, John!');
            expect(user2.greet()).toBe('Hello, Jane!');

            // Method borrowing
            expect(greet.call(user1)).toBe('Hello, John!');
            expect(greet.call(user2)).toBe('Hello, Jane!');
        });
    });
});
```
